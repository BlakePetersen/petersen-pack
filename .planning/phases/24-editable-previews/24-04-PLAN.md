       ```tsx
       export function PlaygroundPropsForm({ props, values, onChange }: PlaygroundPropsFormProps) {
         const visibleProps = props.filter(p =&gt; p.name !== 'children' &amp;&amp; !p.type.includes('=&gt;'))
         if (visibleProps.length === 0) {
           return &lt;p className="text-sm text-muted-foreground py-4"&gt;No props documented&lt;/p&gt;
         }
         const update = (name: string, value: string) =&gt; onChange({ ...values, [name]: value })
         return (
           &lt;div className="grid grid-cols-1 md:grid-cols-2 gap-4"&gt;
             {visibleProps.map(prop =&gt; {
               const control = parsePropType(prop.type)
               const current = values[prop.name] ?? prop.default ?? ''
               switch (control.kind) {
                 case 'boolean':
                   return (
                     &lt;label key={prop.name} className="flex items-center gap-2 font-mono text-xs"&gt;
                       &lt;Toggle pressed={current === 'true'} onPressedChange={v =&gt; update(prop.name, String(v))}&gt;{prop.name}&lt;/Toggle&gt;
                     &lt;/label&gt;
                   )
                 case 'select':
                   return (
                     &lt;label key={prop.name} className="flex flex-col gap-1 font-mono text-xs"&gt;
                       &lt;span&gt;{prop.name}&lt;/span&gt;
                       &lt;select name={prop.name} value={current} onChange={e =&gt; update(prop.name, e.target.value)} className="bg-card border border-border px-2 py-1"&gt;
                         {control.options.map(o =&gt; &lt;option key={o} value={o}&gt;{o}&lt;/option&gt;)}
                       &lt;/select&gt;
                     &lt;/label&gt;
                   )
                 case 'number':
                   return (
                     &lt;label key={prop.name} className="flex flex-col gap-1 font-mono text-xs"&gt;
                       &lt;span&gt;{prop.name}&lt;/span&gt;
                       &lt;Input type="number" name={prop.name} value={current} onChange={e =&gt; update(prop.name, e.target.value)} /&gt;
                     &lt;/label&gt;
                   )
                 case 'text':
                 default:
                   return (
                     &lt;label key={prop.name} className="flex flex-col gap-1 font-mono text-xs"&gt;
                       &lt;span&gt;{prop.name}&lt;/span&gt;
                       &lt;Input type="text" name={prop.name} value={current} onChange={e =&gt; update(prop.name, e.target.value)} /&gt;
                     &lt;/label&gt;
                   )
               }
             })}
           &lt;/div&gt;
         )
       }
       ```
